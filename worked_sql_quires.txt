CREATE TABLE Students (
    student_id INT PRIMARY KEY,
    name VARCHAR(100),
    age INT,
    grade VARCHAR(10)
);

SELECT * FROM Students;


SELECT * FROM Students
WHERE grade = 'A';


UPDATE Students
SET grade = 'A+'
WHERE name = 'Bob';


DELETE FROM Students
WHERE student_id = 3;


SELECT COUNT(*) AS total_students, AVG(age) AS avg_age
FROM Students;


SELECT grade, COUNT(*) AS count
FROM Students
GROUP BY grade
HAVING COUNT(*) > 1;
